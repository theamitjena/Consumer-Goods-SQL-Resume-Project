5.  Get the products that have the highest and lowest manufacturing costs.
The final output should contain these fields,
product_code
product
manufacturing_cost

ANS-

-------------------------------

with ranked_product as (
select 
	p.product_code, p.product
    , c.manufacturing_cost,
    rank () over (order by c.manufacturing_cost desc) as rank_high,
    rank () over (order by c.manufacturing_cost asc) as rank_low
from dim_product p
join fact_manufacturing_cost c
on p.product_code = c.product_code
)
select 
	product_code, product
    , manufacturing_cost
from ranked_product
where rank_high = 1 or rank_low = 1
